cmake_minimum_required(VERSION 3.24)
project(PointyPointer VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
set(EXPAT_MSVC_STATIC_CRT ON CACHE BOOL "" FORCE)
include(CTest)
include(cmake/Dependencies.cmake)
find_package(OpenGL REQUIRED)
add_library(pointy_assets STATIC src/core/sha256.c src/media/assets.c src/media/image.c src/media/gif.c src/media/svg.c src/media/svg_validate.c)
target_sources(pointy_assets PRIVATE src/media/fonts.c src/media/text.c src/media/text_raster.c)
include(cmake/BitmapKernel.cmake)
target_include_directories(pointy_assets PUBLIC include)
target_link_libraries(pointy_assets PRIVATE SDL3_image::SDL3_image-static SDL3::SDL3-static plutosvg::plutosvg expat::expat)
target_link_libraries(pointy_assets PRIVATE SDL3_ttf::SDL3_ttf-static Freetype::Freetype harfbuzz::harfbuzz)
if(NOT MSVC)
  target_link_libraries(pointy_assets PRIVATE m)
endif()
add_library(pointy_display STATIC src/display/layout.c src/display/anchors.c)
target_include_directories(pointy_display PUBLIC include)
if(NOT MSVC)
  target_link_libraries(pointy_display PUBLIC m)
endif()
add_library(pointy_interaction STATIC src/core/geometry.c src/core/interaction.c src/core/easing.c)
target_include_directories(pointy_interaction PUBLIC include)
if(NOT MSVC)
  target_link_libraries(pointy_interaction PUBLIC m)
endif()
add_library(pointy_render STATIC src/render/gl_api.c src/render/frame.c)
target_include_directories(pointy_render PUBLIC include)
target_link_libraries(pointy_render PRIVATE pointy_interaction SDL3::SDL3-static OpenGL::GL)
add_library(pointy_model STATIC src/core/project.c src/core/items.c src/core/scene.c src/media/default_fonts.c)
add_library(pointy_store STATIC src/project/manifest.c src/project/archive.c src/project/store.c)
add_library(pointy_app STATIC src/app/commands.c src/app/local_preferences.c)
target_include_directories(pointy_app PUBLIC include)
target_link_libraries(pointy_app PUBLIC pointy_store PRIVATE SDL3::SDL3-static yyjson)
target_include_directories(pointy_store PUBLIC include)
target_link_libraries(pointy_store PUBLIC pointy_model PRIVATE miniz yyjson)
target_include_directories(pointy_model PUBLIC include)
target_link_libraries(pointy_model PUBLIC pointy_assets pointy_render PRIVATE SDL3::SDL3-static)
if(WIN32)
  configure_file(cmake/windows.manifest.in ${CMAKE_CURRENT_BINARY_DIR}/windows.manifest @ONLY)
  add_library(pointy_platform STATIC src/platform/windows/surface.c src/platform/windows/events.c src/platform/windows/input.c src/platform/windows/displays.c)
  target_sources(pointy_platform PRIVATE src/platform/windows/direct_gl.c)
  target_sources(pointy_platform PRIVATE src/platform/windows/fonts.c)
  target_sources(pointy_platform PRIVATE src/platform/windows/files.c)
  target_sources(pointy_platform PRIVATE src/platform/windows/dialogs.c)
  target_sources(pointy_platform PRIVATE src/platform/windows/tray.c)
  target_link_libraries(pointy_store PRIVATE pointy_platform)
  target_include_directories(pointy_platform PUBLIC include)
  target_link_libraries(pointy_platform PRIVATE user32 gdi32 shcore)
  target_link_libraries(pointy_platform PRIVATE pointy_assets advapi32 SDL3::SDL3-static)
  target_link_libraries(pointy_model PRIVATE pointy_platform)
elseif(APPLE)
  add_library(pointy_platform STATIC src/platform/macos/surface.m src/platform/macos/events.m
    src/platform/macos/input.m src/platform/macos/displays.m src/platform/macos/fonts.m
    src/platform/macos/files.m src/platform/macos/dialogs.m src/platform/macos/tray.m)
  target_include_directories(pointy_platform PUBLIC include)
  target_compile_options(pointy_platform PRIVATE -fno-objc-arc -fblocks)
  target_link_libraries(pointy_platform PRIVATE pointy_assets SDL3::SDL3-static
    "-framework Cocoa" "-framework CoreGraphics" "-framework CoreText")
  target_link_libraries(pointy_model PRIVATE pointy_platform)
  target_link_libraries(pointy_store PRIVATE pointy_platform)
endif()
if(WIN32 OR APPLE)
  add_library(pointy_anchors STATIC src/app/anchors.c src/app/sprites.c)
  add_library(pointy_ui STATIC src/ui/backend.c src/ui/events.c src/ui/theme.c src/ui/settings.c src/ui/item_editor.c src/ui/project_actions.c)
  target_sources(pointy_ui PRIVATE src/ui/text_layout.c src/ui/edit_state.c src/ui/shaped_edit.c)
  target_link_libraries(pointy_ui PRIVATE SDL3_ttf::SDL3_ttf-static)
  target_include_directories(pointy_ui PUBLIC include PRIVATE ${nuklear_SOURCE_DIR})
  target_link_libraries(pointy_ui PUBLIC pointy_app PRIVATE SDL3::SDL3-static OpenGL::GL)
  target_sources(pointy_ui PRIVATE src/ui/drawer.c)
  add_library(pointy_runtime STATIC src/app/runtime.c)
  target_include_directories(pointy_runtime PUBLIC include)
  target_link_libraries(pointy_runtime PUBLIC pointy_anchors pointy_ui)
  add_library(pointy_scheduler STATIC src/app/scheduler.c)
  if(NOT MSVC)
    target_link_libraries(pointy_scheduler PUBLIC m)
  endif()
  add_library(pointy_lifecycle STATIC src/app/lifecycle.c src/app/tray.c)
  target_link_libraries(pointy_lifecycle PUBLIC pointy_runtime pointy_scheduler)
  target_link_libraries(pointy_runtime PRIVATE pointy_scheduler)
  target_link_libraries(pointy_lifecycle PRIVATE SDL3::SDL3-static)
  target_include_directories(pointy_anchors PUBLIC include)
  target_link_libraries(pointy_anchors PUBLIC pointy_display pointy_interaction pointy_render pointy_platform pointy_model PRIVATE SDL3::SDL3-static)
  add_executable(PointyPointer src/main.c src/app/overlay_probe.c src/app/text_probe.c)
  target_link_libraries(PointyPointer PRIVATE pointy_lifecycle SDL3::SDL3-static)
  target_compile_definitions(PointyPointer PRIVATE PP_PROBE_IMAGE="default-hand-proof.bmp")
  if(WIN32)
    set_target_properties(PointyPointer PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(PointyPointer PRIVATE /ENTRY:mainCRTStartup)
    target_sources(PointyPointer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.manifest")
  else()
    set_target_properties(PointyPointer PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
  endif()
endif()
include(cmake/Tests.cmake)
include(cmake/ApplicationResources.cmake)
include(cmake/Packaging.cmake)
