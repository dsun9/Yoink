cmake_minimum_required(VERSION 3.24)
project(PointyPointer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
include(cmake/Dependencies.cmake)
find_package(OpenGL REQUIRED)
add_library(pointy_interaction STATIC src/core/geometry.c src/core/interaction.c src/core/easing.c)
target_include_directories(pointy_interaction PUBLIC include)
if(NOT MSVC)
  target_link_libraries(pointy_interaction PUBLIC m)
endif()
add_library(pointy_render STATIC src/render/gl_api.c src/render/frame.c)
target_include_directories(pointy_render PUBLIC include)
target_link_libraries(pointy_render PRIVATE SDL3::SDL3-static OpenGL::GL)
if(WIN32)
  configure_file(cmake/windows.manifest.in ${CMAKE_CURRENT_BINARY_DIR}/windows.manifest @ONLY)
  add_library(pointy_platform STATIC src/platform/windows/surface.c src/platform/windows/events.c src/platform/windows/input.c)
  target_include_directories(pointy_platform PUBLIC include)
  target_link_libraries(pointy_platform PRIVATE user32 gdi32)
  add_executable(PointyPointer src/main.c src/app/overlay_probe.c)
  target_link_libraries(PointyPointer PRIVATE pointy_render pointy_platform SDL3::SDL3-static)
  target_link_options(PointyPointer PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/windows.manifest")
endif()
include(cmake/Tests.cmake)
